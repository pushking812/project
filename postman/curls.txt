# Наборы curl для проверки API. В Postman: Import -> Raw text -> вставить команду.
# Сервер: npm start (по умолчанию http://127.0.0.1:8737). Для cmd.exe/PowerShell используйте curl.exe и двойные кавычки (см. ниже).

# --- Linux / macOS / Git Bash ---

# 1. health
curl -s http://127.0.0.1:8737/health

# 2. me
curl -s http://127.0.0.1:8737/me

# 3. ask (SSE)
curl -N -X POST http://127.0.0.1:8737/ask -H "Content-Type: application/json" -d '{"prompt":"what is in ASK.md? One line.","useContext":true}'

# 4. execute (SSE)
curl -N -X POST http://127.0.0.1:8737/execute -H "Content-Type: application/json" -d '{"prompt":"create hello.txt with the text Hello","mode":"all"}'

# 5. start (запомните taskId)
curl -s -X POST http://127.0.0.1:8737/start -H "Content-Type: application/json" -d '{"prompt":"create hello.txt with the text Hello","mode":"all"}'

# 6. status (подставьте taskId)
curl -s http://127.0.0.1:8737/status/TASK_ID

# 7. stop
curl -s -X POST http://127.0.0.1:8737/stop/TASK_ID

# --- Ошибки (ожидаемые коды) ---

# 400 пустой prompt
curl -i -X POST http://127.0.0.1:8737/ask -H "Content-Type: application/json" -d '{"prompt":""}'

# 400 неверный mode
curl -i -X POST http://127.0.0.1:8737/execute -H "Content-Type: application/json" -d '{"prompt":"x","mode":"bogus"}'

# 404 неизвестная задача
curl -i http://127.0.0.1:8737/status/no-such-id

# 404 неизвестный маршрут
curl -i http://127.0.0.1:8737/nope

# 405 неверный метод
curl -i -X DELETE http://127.0.0.1:8737/health

# --- Windows (PowerShell / cmd): JSON в двойных кавычках с экранированием ---

curl.exe -s http://127.0.0.1:8737/health
curl.exe -N -X POST http://127.0.0.1:8737/ask -H "Content-Type: application/json" -d "{\"prompt\":\"what is in ASK.md? One line.\",\"useContext\":true}"
curl.exe -N -X POST http://127.0.0.1:8737/execute -H "Content-Type: application/json" -d "{\"prompt\":\"create hello.txt with the text Hello\",\"mode\":\"all\"}"
curl.exe -s -X POST http://127.0.0.1:8737/start -H "Content-Type: application/json" -d "{\"prompt\":\"create hello.txt with the text Hello\",\"mode\":\"all\"}"

# =====================================================================
# ЭТАП 2 (Linux / macOS / Git Bash). Для Windows: curl.exe и JSON в двойных кавычках с \" как выше.
# =====================================================================

# --- 2A: файлы и команды (ключ ASK_KEY не нужен) ---
curl -s -X POST http://127.0.0.1:8737/write -H "Content-Type: application/json" -d '{"path":"hello-postman.txt","content":"Hello\nWorld\n","overwrite":false}'
curl -s -X POST http://127.0.0.1:8737/read -H "Content-Type: application/json" -d '{"path":"hello-postman.txt","offset":1,"limit":20}'
curl -s -X POST http://127.0.0.1:8737/edit -H "Content-Type: application/json" -d '{"path":"hello-postman.txt","old_text":"World","new_text":"Postman"}'
curl -s -X POST http://127.0.0.1:8737/list -H "Content-Type: application/json" -d '{"path":"","depth":1}'
curl -s -X POST http://127.0.0.1:8737/search -H "Content-Type: application/json" -d '{"pattern":"Hello","regex":false}'
curl -s -X POST http://127.0.0.1:8737/command -H "Content-Type: application/json" -d '{"command":"node -v","timeout_sec":20}'

# 403 secret / outside / plan_mode / secret_command
curl -i -X POST http://127.0.0.1:8737/read -H "Content-Type: application/json" -d '{"path":".env"}'
curl -i -X POST http://127.0.0.1:8737/read -H "Content-Type: application/json" -d '{"path":"../../../etc/passwd"}'
curl -i -X POST http://127.0.0.1:8737/write -H "Content-Type: application/json" -d '{"path":"x.txt","content":"x","mode":"plan"}'
curl -i -X POST http://127.0.0.1:8737/command -H "Content-Type: application/json" -d '{"command":"type .env"}'
# 400 нет path
curl -i -X POST http://127.0.0.1:8737/read -H "Content-Type: application/json" -d '{}'

# --- 2B: заметки и память ---
curl -s -X POST http://127.0.0.1:8737/remember -H "Content-Type: application/json" -d '{"note":"postman check: tests run with npm test"}'
curl -s http://127.0.0.1:8737/notes
curl -s -X POST http://127.0.0.1:8737/memory -H "Content-Type: application/json" -d '{"topic":"postman.md","content":"# Postman\nLine one\n"}'
curl -s -X POST http://127.0.0.1:8737/memory -H "Content-Type: application/json" -d '{"topic":"postman.md","old_text":"Line one","new_text":"Line two"}'
curl -s http://127.0.0.1:8737/memory
curl -s http://127.0.0.1:8737/memory/postman.md
# 404 нет темы, 400 плохое имя, 403 очистка без mode=all
curl -i http://127.0.0.1:8737/memory/nope.md
curl -i -X POST http://127.0.0.1:8737/memory -H "Content-Type: application/json" -d '{"topic":"../x.md","content":"x"}'
curl -i -X DELETE "http://127.0.0.1:8737/memory?mode=plan"
# очистка всей памяти папки
curl -s -X DELETE "http://127.0.0.1:8737/memory?mode=all"

# --- 2C: сессии ---
# создать сессию и запомнить id
SID=$(curl -s -X POST http://127.0.0.1:8737/sessions -H "Content-Type: application/json" -d '{"title":"postman demo"}' | node -p "JSON.parse(require('fs').readFileSync(0)).session_id")
echo $SID
curl -s http://127.0.0.1:8737/sessions
# задачи в сессии (нужен ASK_KEY): вторая видит историю первой
curl -N -X POST http://127.0.0.1:8737/execute -H "Content-Type: application/json" -d "{\"prompt\":\"create session-note.txt with the text one\",\"mode\":\"all\",\"session_id\":\"$SID\"}"
curl -N -X POST http://127.0.0.1:8737/execute -H "Content-Type: application/json" -d "{\"prompt\":\"in the file you just made, change one to two\",\"mode\":\"all\",\"session_id\":\"$SID\"}"
# фоновая задача в сессии (подставьте taskId; в /status будет session_id)
curl -s -X POST http://127.0.0.1:8737/start -H "Content-Type: application/json" -d "{\"prompt\":\"what did I ask you before? One line.\",\"mode\":\"plan\",\"session_id\":\"$SID\"}"
curl -s http://127.0.0.1:8737/status/TASK_ID
# просмотр, экспорт в Markdown, resume
curl -s http://127.0.0.1:8737/sessions/$SID
curl -s -X POST http://127.0.0.1:8737/sessions/$SID/export -H "Content-Type: application/json" -d '{"filename":"session-export.md"}'
curl -s -X POST http://127.0.0.1:8737/sessions/resume -H "Content-Type: application/json" -d "{\"id\":\"$SID\"}"
# ошибки: 404 нет сессии, 400 плохой id, 400 плохой session_id в задаче, 403 export в plan, 409 повторный export
curl -i -X POST http://127.0.0.1:8737/sessions/resume -H "Content-Type: application/json" -d '{"id":"no-such-session"}'
curl -i http://127.0.0.1:8737/sessions/no-such-session
curl -i "http://127.0.0.1:8737/sessions/bad%20id"
curl -i -X POST http://127.0.0.1:8737/execute -H "Content-Type: application/json" -d '{"prompt":"x","session_id":"bad id!"}'
curl -i -X POST http://127.0.0.1:8737/sessions/$SID/export -H "Content-Type: application/json" -d '{"filename":"x.md","mode":"plan"}'
